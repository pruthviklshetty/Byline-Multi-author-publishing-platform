{# Reusable presentation helpers. Pure Jinja, no backend changes required. #} {# Estimated reading time from a plain-text body, floored at 1 minute. #} {% macro reading_time(text) -%} {%- set words = (text or "").split() | length -%} {%- set minutes = (words / 200) | round(0, "ceil") | int -%} {{ [minutes, 1] | max }} min read {%- endmacro %} {# A single article row, shared by the homepage and author pages. #} {% macro entry(post, number) -%}
  • {{ "%02d" % number }}

    {{ post.title }}

    {% if post.content %}

    {{ post.content | truncate(170, false, "…") }}

    {% endif %}

    {{ post.author.username }} · {{ post.date_posted.strftime("%b %d, %Y") }} · {{ reading_time(post.content) }}

    • {%- endmacro %}
